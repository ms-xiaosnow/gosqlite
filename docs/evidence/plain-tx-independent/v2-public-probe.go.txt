package sqlite

import (
 "context"
 "errors"
 "testing"
 "github.com/ms-xiaosnow/gosqlite/internal/pager"
)

func TestPlainTxIndependentAutomaticRollbackCodePrecedence(t *testing.T) {
 // Descriptor-free exact composition of DB.Exec's automatic-transaction error
 // branch, now through its production helper: already-classified commit error
 // followed by rollback retirement. Original failing composition is retained.
 db := &DB{rows: make(map[*Rows]struct{})}
 tx := &Tx{db: db, ctx: context.Background(), readClose: func() error {
  return errors.Join(pager.ErrPlaintextCleanupPending, pager.ErrPoisoned)
 }}
 db.tx = tx
 operation := wrap("commit", ErrBusy)
 err := tx.rollbackAfterExecLocked(operation)
 var detail *Error
 if !errors.Is(err, ErrPlaintextCleanupPending) || !errors.As(err, &detail) { t.Fatal(err) }
 if detail.Code != CodeIOErr { t.Fatalf("automatic rollback cleanup masked by earlier operation code %d, want CodeIOErr %d: %v", detail.Code, CodeIOErr, err) }
}
