// Independent expected ordering: exact rational arithmetic instead of the
// production integer/float comparator; byte ordering instead of its helpers.
func reviewClass(v any) int {
	switch x := v.(type) {
	case nil:
		return 0
	case int64:
		return 1
	case float64:
		if math.IsNaN(x) {
			return 0
		}
		return 1
	case string:
		return 2
	case []byte:
		return 3
	}
	panic("test type")
}
func reviewField(a, b any) int {
	x, y := reviewClass(a), reviewClass(b)
	if x < y {
		return -1
	}
	if x > y {
		return 1
	}
	switch x {
	case 0:
		return 0
	case 1:
		af, aok := a.(float64)
		bf, bok := b.(float64)
		if aok && math.IsInf(af, 0) {
			if bok && af == bf {
				return 0
			}
			if math.IsInf(af, -1) {
				return -1
			}
			return 1
		}
		if bok && math.IsInf(bf, 0) {
			if math.IsInf(bf, -1) {
				return 1
			}
			return -1
		}
		ar, br := new(big.Rat), new(big.Rat)
		if aok {
			ar.SetFloat64(af)
		} else {
			ar.SetInt64(a.(int64))
		}
		if bok {
			br.SetFloat64(bf)
		} else {
			br.SetInt64(b.(int64))
		}
		return ar.Cmp(br)
	case 2:
		return bytes.Compare([]byte(a.(string)), []byte(b.(string)))
	case 3:
		return bytes.Compare(a.([]byte), b.([]byte))
	}
	panic("class")
}
func reviewPrefix(a, b []any) int {
	for i := range b {
		if c := reviewField(a[i], b[i]); c != 0 {
			return c
		}
	}
	return 0
}
func reviewMatches(k []any, l, u *IndexBound) bool {
	if l != nil {
		c := reviewPrefix(k, l.Key)
		if c < 0 || c == 0 && !l.Inclusive {
			return false
		}
	}
	if u != nil {
		c := reviewPrefix(k, u.Key)
		if c > 0 || c == 0 && !u.Inclusive {
			return false
		}
	}
	return true
}

// Every tree is constructed from independently sorted keys. Vary the number of
// separators per interior page, rather than only the owner's binary tree shape.
func reviewTree(t *testing.T, keys [][]any, fanout int) (*indexTestReader, uint32) {
	t.Helper()
	r := newIndexTestReader()
	r.next = 100000
	next := uint32(2)
	var build func([][]any, int) uint32
	build = func(part [][]any, depth int) uint32 {
		n := next
		next++
		if depth == 0 {
			addIndexTestPage(t, r, n, true, part, nil)
			return n
		}
		groups := fanout
		sep := make([][]any, 0, groups-1)
		child := make([]uint32, 0, groups)
		at := 0
		for i := 0; i < groups; i++ {
			remain := len(part) - at
			count := (remain - (groups - i - 1)) / (groups - i)
			child = append(child, build(part[at:at+count], depth-1))
			at += count
			if i < groups-1 {
				sep = append(sep, part[at])
				at++
			}
		}
		addIndexTestPage(t, r, n, false, sep, child)
		return n
	}
	return r, build(keys, 2)
}
func TestReviewBoundedIndependentPrefixExhaustive(t *testing.T) {
	vals := []any{nil, math.Inf(-1), int64(math.MinInt64), math.Nextafter(-9223372036854775808.0, math.Inf(-1)), int64(-9007199254740993), float64(-9007199254740992), int64(0), math.Copysign(0, -1), math.SmallestNonzeroFloat64, int64(9007199254740993), float64(9007199254740992), int64(math.MaxInt64), float64(9223372036854775808.0), math.Inf(1), "", "a\x00", "a\x00b", []byte{}, []byte{0}, []byte{0, 255}}
	var keys [][]any
	for i, a := range vals {
		for j, b := range []any{nil, int64(-1), "\x00"} {
			for k := 0; k < 2; k++ {
				id := int64((i*6+j*2+k)*7919 - 400000)
				keys = append(keys, []any{a, b, id})
			}
		}
	}
	sort.Slice(keys, func(i, j int) bool { return reviewPrefix(keys[i], keys[j]) < 0 })
	bounds := []*IndexBound{nil, {[]any{math.NaN()}, true}, {[]any{"not-present"}, false}}
	for _, a := range vals {
		for _, inc := range []bool{false, true} {
			bounds = append(bounds, &IndexBound{[]any{a}, inc})
		}
	}
	rng := rand.New(rand.NewSource(11022026))
	for i := 0; i < 30; i++ {
		k := keys[rng.Intn(len(keys))]
		for _, inc := range []bool{false, true} {
			bounds = append(bounds, &IndexBound{k[:2+i%2], inc})
		}
	}
	cases := 0
	for _, fanout := range []int{3, 4} {
		r, root := reviewTree(t, keys, fanout)
		for _, lo := range bounds {
			for _, hi := range bounds {
				var want [][]any
				for _, k := range keys {
					if reviewMatches(k, lo, hi) {
						want = append(want, k)
					}
				}
				for _, rev := range []bool{false, true} {
					c, e := NewBoundedIndexCursor(context.Background(), r, root, KeyInfo{2}, lo, hi, rev, RangeLimits{})
					if e != nil {
						t.Fatal(e)
					}
					got := collectBoundedIndex(t, c)
					expected := append([][]any(nil), want...)
					if rev {
						for a, b := 0, len(expected)-1; a < b; a, b = a+1, b-1 {
							expected[a], expected[b] = expected[b], expected[a]
						}
					}
					if !reflect.DeepEqual(got, expected) {
						t.Fatalf("fanout %d rev %v lo %#v hi %#v got %v want %v", fanout, rev, lo, hi, got, expected)
					}
					cases++
					r.reads = nil
				}
			}
		}
	}
	t.Logf("independent exact-rational/prefix cases=%d keys=%d", cases, len(keys))
}
func TestReviewBoundedIndependentRowIDBoundaries(t *testing.T) {
	ctx := context.Background()
	p := newMemoryPages(512)
	root, e := CreateTree(ctx, p, TableTree)
	if e != nil {
		t.Fatal(e)
	}
	ids := []int64{math.MinInt64, math.MinInt64 + 1, math.MaxInt64 - 1, math.MaxInt64}
	for i := int64(-80); i <= 80; i++ {
		ids = append(ids, i*97)
	}
	sort.Slice(ids, func(i, j int) bool { return ids[i] < ids[j] })
	for i, id := range ids {
		if e := InsertTable(ctx, p, root, id, []any{fmt.Sprint(id), strings.Repeat("p", i%5*180)}); e != nil {
			t.Fatal(e)
		}
	}
	limits := []*RowIDBound{nil}
	for _, id := range ids {
		for _, inc := range []bool{false, true} {
			limits = append(limits, &RowIDBound{id, inc})
		}
	}
	cases := 0
	rng := rand.New(rand.NewSource(777))
	for n := 0; n < 2500; n++ {
		lo, hi := limits[rng.Intn(len(limits))], limits[rng.Intn(len(limits))]
		var want []int64
		for _, id := range ids {
			if lo != nil && (id < lo.Value || id == lo.Value && !lo.Inclusive) || hi != nil && (id > hi.Value || id == hi.Value && !hi.Inclusive) {
				continue
			}
			want = append(want, id)
		}
		for _, rev := range []bool{false, true} {
			c, e := NewRowIDRangeCursor(ctx, p, root, lo, hi, rev, RangeLimits{})
			if e != nil {
				t.Fatal(e)
			}
			var got []int64
			for {
				id, v, e := c.Next()
				if e == io.EOF {
					break
				}
				if e != nil {
					t.Fatal(e)
				}
				if v[0] != fmt.Sprint(id) {
					t.Fatal("row payload", id, v)
				}
				got = append(got, id)
			}
			expected := append([]int64(nil), want...)
			if rev {
				for a, b := 0, len(expected)-1; a < b; a, b = a+1, b-1 {
					expected[a], expected[b] = expected[b], expected[a]
				}
			}
			if !reflect.DeepEqual(got, expected) {
				t.Fatal(lo, hi, rev, got, expected)
			}
			c.Close()
			cases++
		}
	}
	t.Logf("independent rowid boundary cases=%d records=%d", cases, len(ids))
}
func TestReviewBoundedIndependentInteriorSelectivity(t *testing.T) {
	var keys [][]any
	for i := 0; i < 120; i++ {
		keys = append(keys, []any{fmt.Sprintf("%03d", i), int64(i - 60)})
	}
	r, root := reviewTree(t, keys, 4)
	full, e := NewIndexCursor(context.Background(), r, root, KeyInfo{1})
	if e != nil {
		t.Fatal(e)
	}
	for {
		_, e := full.Next()
		if e == io.EOF {
			break
		}
		if e != nil {
			t.Fatal(e)
		}
	}
	full.Close()
	allReads := len(r.reads)
	for _, rev := range []bool{false, true} {
		for _, at := range []int{0, 29, 30, 59, 60, 89, 90, 119} {
			r.reads = nil
			bound := &IndexBound{keys[at], true}
			c, e := NewBoundedIndexCursor(context.Background(), r, root, KeyInfo{1}, bound, bound, rev, RangeLimits{})
			if e != nil {
				t.Fatal(e)
			}
			got := collectBoundedIndex(t, c)
			if !reflect.DeepEqual(got, keys[at:at+1]) || len(r.reads) > 3 || len(r.reads) >= allReads {
				t.Fatal(at, rev, got, r.reads, allReads)
			}
			t.Logf("at=%d reverse=%v exact trace=%v complete scan=%d", at, rev, r.reads, allReads)
		}
	}
}
func TestReviewBoundedIndependentMemoryParserRefusal(t *testing.T) {
	for _, rev := range []bool{false, true} {
		r := newIndexTestReader()
		addIndexTestPage(t, r, 2, false, [][]any{{int64(0), int64(0)}}, []uint32{3, 4})
		addIndexTestPage(t, r, 3, true, [][]any{{int64(1), int64(-1)}}, nil)
		addIndexTestPage(t, r, 4, true, [][]any{{int64(-1), int64(1)}}, nil)
		c, e := NewBoundedIndexCursor(context.Background(), r, 2, KeyInfo{1}, nil, nil, rev, RangeLimits{})
		if e == nil {
			_, e = c.Next()
			c.Close()
		}
		if !errors.Is(e, pager.ErrCorrupt) {
			t.Fatal("memory parent bound", rev, e)
		}
	}
	r := newIndexTestReader()
	addIndexTestPage(t, r, 2, true, [][]any{{"a", int64(1)}}, nil)
	binary.BigEndian.PutUint16(r.pages[2][3:5], 65535)
	if _, e := NewBoundedIndexCursor(context.Background(), r, 2, KeyInfo{1}, nil, nil, false, RangeLimits{}); !errors.Is(e, pager.ErrCorrupt) {
		t.Fatal("memory cell count", e)
	}
	b, e := newRangeBudget(context.Background(), r, RangeLimits{PageReads: math.MaxInt64, Work: math.MaxInt64, Records: math.MaxInt64, PayloadBytes: math.MaxInt64})
	if e != nil {
		t.Fatal(e)
	}
	b.stats.Work = math.MaxInt64 - 1
	if e = b.work(1); e != nil {
		t.Fatal(e)
	}
	if e = b.work(1); !errors.Is(e, record.ErrLimit) || b.stats.Work != math.MaxInt64 {
		t.Fatal("overflow", e, b.stats)
	}
}

// The upstream admission hook may observe cancellation while it does its own
// reservation checks. Forwarding must recheck it before page-parser allocation.
type reviewCancelAdmissionReader struct {
	PageReader
	cancel context.CancelFunc
}

func (r reviewCancelAdmissionReader) BeforePage(int64) error { r.cancel(); return nil }
func TestReviewBoundedIndependentCancelDuringAdmission(t *testing.T) {
	ctx, cancel := context.WithCancel(context.Background())
	defer cancel()
	b, e := newRangeBudget(ctx, reviewCancelAdmissionReader{cancel: cancel}, RangeLimits{})
	if e != nil {
		t.Fatal(e)
	}
	if e = b.BeforePage(512); !errors.Is(e, context.Canceled) {
		t.Fatalf("page admission must return context cancellation before parser allocation, got %v", e)
	}
}

func TestReviewBoundedIndependentEndpointAdmission(t *testing.T) {
	r := newIndexTestReader()
	addIndexTestPage(t, r, 2, true, [][]any{{"a", int64(1)}}, nil)
	lo := &IndexBound{[]any{bytes.Repeat([]byte{0}, 5000)}, true}
	hi := &IndexBound{[]any{bytes.Repeat([]byte{255}, 5000)}, true}
	_, e := NewBoundedIndexCursor(context.Background(), r, 2, KeyInfo{1}, lo, hi, false, RangeLimits{Work: 6000})
	if !errors.Is(e, record.ErrLimit) || len(r.reads) != 0 {
		t.Fatalf("both bound clones must be admitted before copying or reading: err=%v reads=%v", e, r.reads)
	}
}

func TestReviewBoundedIndependentCumulativeStats(t *testing.T) {
	r := newIndexTestReader()
	keys := [][]any{{"a", int64(-1)}, {"a", int64(0)}, {"a", int64(1)}}
	addIndexTestPage(t, r, 2, true, keys, nil)
	for _, rev := range []bool{false, true} {
		c, e := NewBoundedIndexCursor(context.Background(), r, 2, KeyInfo{1}, nil, nil, rev, RangeLimits{Records: 2})
		if e != nil {
			t.Fatal(e)
		}
		for range 2 {
			if _, e = c.Next(); e != nil {
				t.Fatal(e)
			}
		}
		if _, e = c.Next(); !errors.Is(e, record.ErrLimit) {
			t.Fatal(e)
		}
		s := c.Stats()
		if s.Records != 2 || s.Returned != 2 || s.PageReads != 1 {
			t.Fatal(s)
		}
		if _, again := c.Next(); again != e || c.Stats() != s {
			t.Fatal("sticky stats", e, again, s, c.Stats())
		}
		c.Close()
		if _, e = c.Next(); e != io.EOF || c.Stats() != s {
			t.Fatal(e, s, c.Stats())
		}
	}
	for _, limits := range []RangeLimits{{PageReads: -1}, {Work: -1}, {Records: -1}, {PayloadBytes: -1}} {
		r.reads = nil
		if _, e := NewBoundedIndexCursor(context.Background(), r, 2, KeyInfo{1}, nil, nil, false, limits); !errors.Is(e, record.ErrUnsupported) || len(r.reads) != 0 {
			t.Fatal(limits, e, r.reads)
		}
		if _, e := NewRowIDRangeCursor(context.Background(), r, 2, nil, nil, false, limits); !errors.Is(e, record.ErrUnsupported) || len(r.reads) != 0 {
			t.Fatal(limits, e, r.reads)
		}
	}
}
