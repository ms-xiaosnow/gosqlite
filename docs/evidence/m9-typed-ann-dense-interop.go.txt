package main
import("context";"crypto/sha256";"encoding/hex";"encoding/json";"errors";"fmt";"os";"github.com/ms-xiaosnow/gosqlite/sqlite")
func run() (err error){
 if len(os.Args)!=4{return fmt.Errorf("usage: source_label mode db")};ctx:=context.Background();db,err:=sqlite.Open(os.Args[3],sqlite.Options{});if err!=nil{return err};defer func(){err=errors.Join(err,db.Close())}()
 if os.Args[2]=="create"{if _,err=db.Exec(ctx,"CREATE TABLE items(id INTEGER PRIMARY KEY,embedding VECTOR(3) NOT NULL)");err!=nil{return err};for i:=1;i<=32;i++{if _,err=db.Exec(ctx,"INSERT INTO items VALUES(?,?)",i,fmt.Sprintf("[%d,%d,1]",i,i%7));err!=nil{return err}};if err=db.CreateHNSWIndex(ctx,"h","items","embedding");err!=nil{return err};if err=db.CreateIVFFlatIndex(ctx,"i","items","embedding",sqlite.IVFFlatIndexOptions{Lists:3});err!=nil{return err}}
 for _,name:=range []string{"h","i"}{if err=db.CheckExtensionIndex(ctx,name);err!=nil{return err};r,e:=db.VectorSearch(ctx,"items","embedding","[10,3,1]",sqlite.VectorSearchOptions{Mode:sqlite.VectorModeApproximate,Index:name,Limit:3});if e!=nil{return e};if r.Plan.Exact||r.Plan.MethodVersion!=1||len(r.Hits)!=3{return fmt.Errorf("incorrect legacy ANN identity: %+v",r.Plan)}}

}
