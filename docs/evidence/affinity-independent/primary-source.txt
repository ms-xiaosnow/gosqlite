   369	** pExpr is a comparison operator.  Return the type affinity that should

   372	static char comparisonAffinity(const Expr *pExpr){
   373	  char aff;
   374	  assert( pExpr->op==TK_EQ || pExpr->op==TK_IN || pExpr->op==TK_LT ||
   375	          pExpr->op==TK_GT || pExpr->op==TK_GE || pExpr->op==TK_LE ||
   376	          pExpr->op==TK_NE || pExpr->op==TK_IS || pExpr->op==TK_ISNOT );
   377	  assert( pExpr->pLeft );
   378	  aff = sqlite3ExprAffinity(pExpr->pLeft);
   379	  if( pExpr->pRight ){
   380	    aff = sqlite3CompareAffinity(pExpr->pRight, aff);
   381	  }else if( ExprUseXSelect(pExpr) ){
   382	    aff = sqlite3CompareAffinity(pExpr->x.pSelect->pEList->a[0].pExpr, aff);

   384	    aff = SQLITE_AFF_BLOB;
   385	  }
   386	  return aff;

  5002	/*
  5003	** The WHERE-clause constant propagation optimization.
  5004	**
  5005	** If the WHERE clause contains terms of the form COLUMN=CONSTANT or
  5006	** CONSTANT=COLUMN that are top-level AND-connected terms that are not
  5007	** part of a ON clause from a LEFT JOIN, then throughout the query
  5008	** replace all other occurrences of COLUMN with CONSTANT.
  5009	**
  5010	** For example, the query:
  5011	**
  5012	**      SELECT * FROM t1, t2, t3 WHERE t1.a=39 AND t2.b=t1.a AND t3.c=t2.b
  5013	**
  5014	** Is transformed into
  5015	**
  5016	**      SELECT * FROM t1, t2, t3 WHERE t1.a=39 AND t2.b=39 AND t3.c=39
  5017	**
  5018	** Return true if any transformations where made and false if not.
  5019	**
  5020	** Implementation note:  Constant propagation is tricky due to affinity
  5021	** and collating sequence interactions.  Consider this example:
  5022	**
  5023	**    CREATE TABLE t1(a INT,b TEXT);
  5024	**    INSERT INTO t1 VALUES(123,'0123');
  5025	**    SELECT * FROM t1 WHERE a=123 AND b=a;
  5026	**    SELECT * FROM t1 WHERE a=123 AND b=123;
  5027	**
  5028	** The two SELECT statements above should return different answers.  b=a
  5029	** is always true because the comparison uses numeric affinity, but b=123
  5030	** is false because it uses text affinity and '0123' is not the same as '123'.
  5031	** To work around this, the expression tree is not actually changed from
  5032	** "b=a" to "b=123" but rather the "a" in "b=a" is tagged with EP_FixedCol
  5033	** and the "123" value is hung off of the pLeft pointer.  Code generator
  5034	** routines know to generate the constant "123" instead of looking up the
  5035	** column value.  Also, to avoid collation problems, this optimization is
  5036	** only attempted if the "a=123" term uses the default BINARY collation.
  5037	**
  5038	** 2021-05-25 forum post 6a06202608: Another troublesome case is...
  5039	**
  5040	**    CREATE TABLE t1(x);
  5041	**    INSERT INTO t1 VALUES(10.0);
  5042	**    SELECT 1 FROM t1 WHERE x=10 AND x LIKE 10;
  5043	**
  5044	** The query should return no rows, because the t1.x value is '10.0' not '10'
  5045	** and '10.0' is not LIKE '10'.  But if we are not careful, the first WHERE
  5046	** term "x=10" will cause the second WHERE term to become "10 LIKE 10",
  5047	** resulting in a false positive.  To avoid this, constant propagation for
  5048	** columns with BLOB affinity is only allowed if the constant is used with
  5049	** operators ==, <=, <, >=, >, or IS in a way that will cause the correct
  5050	** type conversions to occur.  See logic associated with the bHasAffBlob flag
  5051	** for details.
  5052	*/
