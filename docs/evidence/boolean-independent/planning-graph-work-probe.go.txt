package sqlite_test

import (
	"context"
	"fmt"
	"path/filepath"
	"strings"
	"testing"
	"time"

	"github.com/ms-xiaosnow/gosqlite/sqlite"
)

// Ordinary empty-table planning diagnostic only; no giant/unbounded input,
// process timeout, termination, or injected filesystem behavior.
func TestBooleanIndependentPlanningGraphWork(t *testing.T) {
	db, e := sqlite.Open(filepath.Join(t.TempDir(), "planning.db"), sqlite.Options{})
	if e != nil {
		t.Fatal(e)
	}
	defer db.Close()
	cols := []string{}
	for i := 0; i <= 18; i++ {
		cols = append(cols, fmt.Sprintf("v%d INTEGER", i))
	}
	for _, q := range []string{"CREATE TABLE c(" + strings.Join(cols, ",") + ")", "CREATE TABLE u(y INTEGER)"} {
		if _, e := db.Exec(context.Background(), q); e != nil {
			t.Fatal(e)
		}
	}
	for _, depth := range []int{4, 8, 12, 16, 18} {
		terms := []string{"v0=1"}
		for i := 1; i <= depth; i++ {
			terms = append(terms, fmt.Sprintf("v%d=v%d+v%d", i, i-1, i-1))
		}
		ctx := sqlite.WithRelationalLimits(context.Background(), sqlite.RelationalLimits{MaxWork: 1000000})
		start := time.Now()
		rows, e := db.Query(ctx, "SELECT v0 FROM c,u WHERE "+strings.Join(terms, " AND ")+" LIMIT 0")
		elapsed := time.Since(start)
		if e != nil {
			t.Fatal(e)
		}
		stats := rows.RelationalStats()
		rows.Close()
		t.Logf("depth=%d query-duration=%s accounted-work=%d source-opens=%d", depth, elapsed, stats.Work, stats.SourceOpens)
	}
}
