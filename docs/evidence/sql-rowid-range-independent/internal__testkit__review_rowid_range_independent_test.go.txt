//go:build oracle

package testkit_test

import (
	"context"
	"fmt"
	"math/rand"
	"os/exec"
	"path/filepath"
	"strings"
	"testing"

	"github.com/ms-xiaosnow/gosqlite/sqlite"
)

func TestReviewRowIDRangeIndependentSQLite354(t *testing.T) {
	confOracle(t)
	for _, size := range []uint32{512, 4096, 65536} {
		t.Run(fmt.Sprint(size), func(t *testing.T) {
			path := filepath.Join(t.TempDir(), "independent.db")
			db, err := sqlite.Open(path, sqlite.Options{PageSize: size})
			if err != nil {
				t.Fatal(err)
			}
			defer db.Close()
			writeExec(t, db, "CREATE TABLE t(id INTEGER PRIMARY KEY,k INTEGER,rowid TEXT,v BLOB)")
			ids := []string{"-9223372036854775808", "-9223372036854775807", "-9007199254740993", "-9007199254740992", "9007199254740992", "9007199254740993", "9223372036854775806", "9223372036854775807"}
			for i := -20; i <= 20; i++ {
				ids = append(ids, fmt.Sprint(i))
			}
			tx, err := db.Begin(context.Background())
			if err != nil {
				t.Fatal(err)
			}
			for n, id := range ids {
				writeTxExec(t, tx, fmt.Sprintf("INSERT INTO t VALUES(%s,%d,'shadow',X'00FF')", id, n%3))
			}
			if err = tx.Commit(context.Background()); err != nil {
				t.Fatal(err)
			}
			before := confDigest(t, path)
			bounds := []string{"-9223372036854775808", "-9223372036854775807", "-9223372036854777856.0", "-9223372036854775808.0", "-9223372036854774784.0", "-9007199254740993", "-0.5", "-0.0", "0.0", "4.9406564584124654e-324", "0.5", "3", "9007199254740993", "9223372036854774784.0", "9223372036854775807", "9223372036854775808.0", "'-9223372036854775808'", "'9223372036854775807'", "'9223372036854775808'", "'1e1'", "'bad'", "X'31'", "X''", "NULL", "-1e999", "1e999"}
			rng := rand.New(rand.NewSource(77101))
			ops := []string{"<", "<=", ">", ">="}
			cases := 0
			for n := 0; n < 384; n++ {
				a, b := bounds[rng.Intn(len(bounds))], bounds[rng.Intn(len(bounds))]
				oa, ob := ops[rng.Intn(4)], ops[rng.Intn(4)]
				where := fmt.Sprintf("id%s%s AND %s%s_rowid_", oa, a, b, ob)
				if n%4 == 0 {
					where += " AND (k<2 OR k IS NULL)"
				}
				direction := []string{"ASC", "DESC"}[n%2]
				order := []string{"id", "1", "rid"}[n%3]
				q := fmt.Sprintf("SELECT id AS rid,k,v FROM t WHERE %s ORDER BY %s %s NULLS %s LIMIT %d OFFSET %d", where, order, direction, []string{"FIRST", "LAST"}[n%2], []int{-1, 0, 1, 3, 7}[n%5], n%6)
				t.Run(fmt.Sprint(n), func(t *testing.T) { writeSame(t, db, path, q) })
				cases++
			}
			for _, q := range []string{
				"SELECT id AS \"true\" FROM t WHERE true>0 ORDER BY true DESC LIMIT 4 OFFSET 1",
				"SELECT id AS \"false\" FROM t WHERE false<0 ORDER BY false LIMIT 4 OFFSET 1",
				"SELECT id AS rowid FROM t WHERE id>0 ORDER BY rowid DESC LIMIT 4",
				"SELECT rowid AS id,_rowid_ FROM t WHERE _rowid_>0 ORDER BY id DESC LIMIT 4",
				"SELECT id,k FROM t WHERE id>-10 AND id<10 ORDER BY k DESC LIMIT 4 OFFSET 1",
				"SELECT id,k FROM t WHERE id>-10 AND id<10 ORDER BY k DESC,id DESC LIMIT 4 OFFSET 1",
				"SELECT id/0,id,v FROM t WHERE id>-10 AND id<10 ORDER BY 2 DESC LIMIT 4 OFFSET 1",
				"SELECT id,k FROM t WHERE +id>'1' ORDER BY id DESC LIMIT 4",
				"SELECT id,k FROM t WHERE id>0 OR id<-10 ORDER BY id DESC LIMIT 4",
				"SELECT t.id FROM t WHERE t.id>0 ORDER BY t.id DESC LIMIT 4",
				"SELECT x.id FROM t AS x WHERE x.id>0 ORDER BY x.id DESC LIMIT 4",
				"SELECT id,k FROM t WHERE id>-10 AND id<10 ORDER BY id DESC LIMIT 4 OFFSET -1",
			} {
				t.Run(fmt.Sprintf("shape_%d", cases), func(t *testing.T) { writeSame(t, db, path, q) })
				cases++
			}
			// Use storage-class-preserving bound parameters, without SQL substitution.
			for n, p := range []struct {
				native any
				sql    string
			}{{int64(-9223372036854775807), "-9223372036854775807"}, {float64(-9223372036854775808.0), "-9223372036854775808.0"}, {float64(9223372036854775808.0), "9223372036854775808.0"}, {"9223372036854775807", "'9223372036854775807'"}, {[]byte("42"), "X'3432'"}, {nil, "NULL"}} {
				q := "SELECT id,k,v FROM t WHERE -?1<id AND id<=?1 ORDER BY id DESC LIMIT 4 OFFSET 1"
				t.Run(fmt.Sprintf("parameter_%d", n), func(t *testing.T) {
					cmd := exec.Command(confOracle(t), "-batch", "-bail", "-quote", "-header", "-readonly", path)
					cmd.Stdin = strings.NewReader(fmt.Sprintf(".parameter init\n.parameter set ?1 %q\n%s;\n", p.sql, q))
					output, err := cmd.CombinedOutput()
					if err != nil {
						t.Fatal(err, string(output))
					}
					parsed, err := confQuoted(output)
					if err != nil {
						t.Fatal(err)
					}
					var want confResult
					if len(parsed) > 0 {
						want.rows = parsed[1:]
						for _, v := range parsed[0] {
							want.columns = append(want.columns, v.(string))
						}
					}
					confCompare(t, want, confNative(t, db, q, p.native))
				})
				cases++
			}
			writeIntegrity(t, path)
			if confDigest(t, path) != before {
				t.Fatal("read-only oracle/query changed intact database bytes")
			}
			t.Logf("%d independent official SQLite comparisons; unchanged ordinary database", cases)
		})
	}
}
