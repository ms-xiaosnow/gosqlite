// Independent ordinary SQL differential probes. All CLI operations finish
// normally; there is no physical mutation or failure injection.
func TestGroupedReviewerDifferential(t *testing.T) {
 db,path:=aggregateOracleFixture(t,false)
 confCLI(t,path,`CREATE TABLE rg(id INTEGER PRIMARY KEY,k,v,w);
INSERT INTO rg VALUES
(1,1.0,4,1),(2,1,1,4),(3,1.0,2,2),(4,1,NULL,NULL),
(5,2,3,9),(6,2.0,9,3),(7,2,3,NULL),(8,2.0,NULL,9),
(9,NULL,NULL,2),(10,NULL,NULL,1),(11,0.0,NULL,NULL),(12,0,NULL,NULL),
(13,'1',7,2),(14,x'31',8,3),(15,'',9,4),(16,x'',10,5),
(17,CAST(x'610062' AS TEXT),11,6),(18,CAST(x'610063' AS TEXT),12,7),
(19,x'610062',13,8),(20,x'610063',14,9),
(21,9007199254740992,2,3),(22,9007199254740992.0,3,2),(23,9007199254740993,4,1),
(24,9223372036854775807,5,6),(25,9223372036854775808.0,6,5),
(26,-9223372036854775808,7,8),(27,-9223372036854775808.0,8,7),
(28,-9007199254740993,9,10),(29,-9007199254740992.0,10,9);
CREATE TABLE re(k,v); INSERT INTO re VALUES(1,1),(2,-9223372036854775808),(3,3);
CREATE TABLE ro(k,v); INSERT INTO ro VALUES(1,1),(2,9223372036854775807),(2,1),(3,3);
CREATE TABLE rz(k,v);`,false)
 var qs []string
 keys:=[]string{"k","+k","k+0","CASE WHEN k IS NULL THEN 8 ELSE k END","CAST(k AS NUMERIC)","CAST(k AS TEXT)","CAST(k AS BLOB)"}
 calls:=[]string{"count(*)","min(v)","max(v)","min(v),max(w)","max(v),min(w)","min(v),max(w),min(v)","min(v),min(w),max(v)","max(w),min(v),max(w)","CASE WHEN count(*)>1 THEN min(v) ELSE max(w) END"}
 for _,key:=range keys { for _,call:=range calls {
  for _,suffix:=range []string{""," HAVING count(*)>0"," ORDER BY 3 DESC,1,2"} {
   qs=append(qs,fmt.Sprintf("SELECT typeof(%s),hex(%s),%s FROM rg GROUP BY %s%s",key,key,call,key,suffix))
  }
 }}
 for _,key:=range []string{"k","+k","k+0","CAST(k AS NUMERIC)"} {
  for _,pred:=range []string{"typeof(%s)='integer'","typeof(%s)='real'","typeof(%s)='text'","typeof(%s)='blob'","(%s) IS NULL","(%s)=1","(%s)='1'","NOT ((%s) IS NULL)","CASE WHEN typeof(%s)='real' THEN 1 ELSE 0 END"} {
   pred=fmt.Sprintf(pred,key)
   for _,having:=range []string{pred,"count(*)>0 AND "+pred,pred+" AND count(*)>0","count(*)<0 OR "+pred} {
    qs=append(qs,fmt.Sprintf("SELECT typeof(%s),hex(%s),count(*),min(v),max(w) FROM rg GROUP BY %s HAVING %s",key,key,key,having))
   }
  }
 }
 for _,q:=range []string{
  "SELECT k,k+0,typeof(k),typeof(k+0),min(v),max(w) FROM rg WHERE id<17 GROUP BY k,k+0",
  "SELECT k+0,k,typeof(k+0),typeof(k),min(v),max(w) FROM rg WHERE id<17 GROUP BY k+0,k",
  "SELECT k+0,typeof(k+0),min(v),max(w) FROM rg WHERE id<17 GROUP BY k+0 ORDER BY k+0 DESC",
  "SELECT count(*) FROM rg GROUP BY (1=1) HAVING count(*)>0",
  "SELECT true,count(*) FROM rg GROUP BY true",
  "SELECT false,count(*) FROM rg GROUP BY false",
  "SELECT k AS q,count(*) AS q FROM rg WHERE id<17 GROUP BY q ORDER BY q",
  "SELECT k AS q,count(*) FROM rg WHERE id<17 GROUP BY +1 ORDER BY -(-1)",
  "SELECT count(*) FROM rg HAVING 0",
  "SELECT count(*) FROM rz HAVING count(*)=0",
 } {qs=append(qs,q)}
 for _,tab:=range []string{"re","ro","rz"} {
  agg:="sum(abs(v))"; if tab=="ro" {agg="sum(v)"}
  for _,having:=range []string{""," HAVING 0"," HAVING 0.0"," HAVING '0'"," HAVING NULL"," HAVING false"," HAVING k=1"," HAVING count(*)=0"," HAVING count(*)>0 AND k=1"," HAVING k=1 OR count(*)<0"} {
   for _,tail:=range []string{""," LIMIT 1"," ORDER BY k DESC LIMIT 1"," ORDER BY k+0 DESC LIMIT 1"," LIMIT 0 OFFSET NULL"} {
    qs=append(qs,"SELECT k,"+agg+" FROM "+tab+" GROUP BY k"+having+tail)
   }
  }
 }
 for _,tail:=range []string{" ORDER BY k+0 LIMIT 1", " ORDER BY k+0 LIMIT 1 OFFSET 1", " ORDER BY k+0 DESC LIMIT 1", " ORDER BY k+0 NULLS LAST LIMIT 1", " ORDER BY k+0,k LIMIT 1"} {
  for _,bad:=range []string{"1","2","3"} {
   qs=append(qs,"SELECT k,CASE WHEN k="+bad+" THEN abs(-9223372036854775808) ELSE count(*) END FROM re GROUP BY k"+tail)
  }
 }
 for _,q:=range qs { t.Run(q,func(t *testing.T){ groupedReviewerCompare(t,db,path,q) }) }
}
func groupedReviewerCompare(t *testing.T,db *sqlite.DB,path,q string) {
 t.Helper()
 cmd:=exec.Command(confOracle(t),"-batch","-bail","-readonly","-quote","-header",path,q)
 raw,oracleErr:=cmd.CombinedOutput()
 if oracleErr==nil {confCompare(t,confExpected(t,path,q),confNative(t,db,q)); return}
 r,err:=db.Query(context.Background(),q)
 var values [][]any
 if err==nil {for r.Next(){values=append(values,r.Values())}; err=r.Err(); r.Close()}
 part:="integer overflow"
 if !strings.Contains(string(raw),part) {t.Fatalf("unclassified official error: %v %s",oracleErr,raw)}
 if err==nil || !strings.Contains(err.Error(),part) {t.Fatalf("oracle error %s; native values %#v error %v",raw,values,err)}
 if len(values)>0 {t.Fatalf("grouped failure leaked partial results %#v, error %v",values,err)}
}
